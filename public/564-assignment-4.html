stp x29, x30, [sp,-16]!decrement sp by 16 first, then store x29 ⇐ sp, x30 ⇐ sp + 8
mov x0,0x0 ⇐ 0
add x29, sp, 0x29 ⇐ x29+0 = x29
bl timebranch to label time
mov x1, x0x1 ⇐ x0 = 0
ldp x29, x30, [sp],16this is post index load pair of pointer: sp ⇐ x29, sp + 8 ⇐ x30, and then increment sp ⇐ sp + 16
adrp x0, .LC0adrp is address of page. this instruction takes the address of .LC0, rounds it down to a 4KB page, loads that page address into x0
add x0,x0, :lo12:.LC0:lo12 means take the lower 12 bits of the symbol address. x0 ⇐ x0 + the lower 12 bits of symbol address .LC0
b printfbranch to the label printf
.LC0:
.string "%d\n"

why split into page + offset adrp can encode large pc-relative distances add handles the 12-bit offset this works with PIE (position independent executables)
